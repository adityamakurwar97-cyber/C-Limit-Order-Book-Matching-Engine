cmake_minimum_required(VERSION 3.16)
project(ExchangeLab VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(exchange src/main.cpp)
add_executable(test_engine tests/test_engine.cpp)
add_executable(benchmark src/benchmark.cpp)
foreach(target exchange test_engine benchmark)
  target_include_directories(${target} PRIVATE include tests)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME correctness COMMAND test_engine)
